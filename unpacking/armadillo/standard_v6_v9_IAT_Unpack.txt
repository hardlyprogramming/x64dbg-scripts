////////////////////////////////////////////////////////////////////////////////
// RFIND - Armadillo IAT-redirect fixer + CRC/IAT-wait detector + OEP finder
//
// Phase 1  : reach the loader's import-resolution code
// Phase 1a : poll across LoadLibraryA calls until the redirect stubs exist,
//            then breakpoint each stub's conditional jump
// Phase 1b : run until the first stub jne fires -> that is the live redirect
// Phase 1c : find every CRC / IAT-wait guard, breakpoint each one
// Phase 1d : dispatch loop -
//              - stopped at the redirect jne -> force ZF=1 so it falls through
//              - stopped at a CRC guard      -> step the test, read ZF;
//                                               ZF == 1 means the IAT is written
//            (if no CRC guard exists, fall back to CreateThread as the signal)
// Phase 1e : IAT complete -> keep only CreateThread, run on to it
// Phase 2  : walk from CreateThread to the original entry point (OEP)
//
// Build notes:
//   - Only je / jne / jz / jnz / jmp are accepted for branching.
//   - Flags are written directly (_ZF = 1) and read directly (cmp _zf, 1);
//     the case of each is kept as tested in the original scripts.
//   - Hex literals are prefixed with 0x.
////////////////////////////////////////////////////////////////////////////////


//============================================================================
// Phase 1 - reach the loader's import-resolution code
//============================================================================
bp VirtualProtect
erun
bc


//============================================================================
// Phase 1a - poll until the redirect stubs exist, then breakpoint each jne
//============================================================================
// Each redirect stub looks like:
//   E9 ?? ?? ?? ??          jmp <redirect>
//   83 B? ?? ?? ?? ?? 00     cmp dword ptr [reg+disp], 0
//   75 44                   jne <take redirect>   <- jne sits 0xC past the E9
//
// The trailing "75 44" separates a real stub from a coincidental byte match.
//
// One LoadLibraryA is often too early: Armadillo builds the stubs as imports
// resolve across several library loads, so a scan right after the first load
// can come back empty. On each miss, run on to the next LoadLibraryA and scan
// again. scan_attempts is only a runaway guard against looping forever.
bp LoadLibraryA
scan_attempts = 0

find_stubs:
erun
rtr
sti
cmp scan_attempts, 200
je no_stubs_found
scan_attempts = scan_attempts + 0x1
findall mem.base(cip), "E9 ?? ?? ?? ?? 83 B? ?? ?? ?? ?? 00 75 44"
cmp $result, 0
je find_stubs

// stubs found -> drop LoadLibraryA and breakpoint every candidate jne
bc
stub_count = $result
idx = 0

arm_stub_bps:
cmp idx, stub_count
je run_to_live_stub
match_addr = ref.addr(idx)
jne_addr = match_addr + 0xC
bp jne_addr
idx = idx + 0x1
jmp arm_stub_bps


//============================================================================
// Phase 1b - lock onto the live redirect jne
//============================================================================
run_to_live_stub:
erun
redirect_jne = eip
log "Live redirect jne @ {p:redirect_jne}"

// Keep only the live jne; drop the other candidates.
bc
bp redirect_jne


//============================================================================
// Phase 1c - find every CRC / IAT-wait guard and breakpoint it
//============================================================================
// Guard shape:
//   85 ??                   test reg, reg     <- breakpoint here (start of match)
//   75 05                   jne +5
//   E9 ?? ?? ?? ??          jmp <keep waiting>
findall mem.base(cip), "85 ?? 75 05 E9"
cmp $result, 0
je no_crc_guard            // no guard found -> use CreateThread as the signal

crc_count = $result
idx = 0
crc_locked = 0

arm_crc_bps:
cmp idx, crc_count
je iat_dispatch
crc_addr = ref.addr(idx)
bp crc_addr
idx = idx + 0x1
jmp arm_crc_bps


//============================================================================
// Phase 1d - dispatch: force redirects, watch the CRC guard for "IAT written"
//============================================================================
// On entry we are stopped at the live redirect jne (from Phase 1b).
// Only two kinds of breakpoint are live here: the redirect jne, and CRC guards.
//crc_locked = 0

iat_dispatch:
cmp eip, redirect_jne
jne crc_hit

// --- redirect jne: force the compare to look "equal" so the jne falls through.
//     ZF is written AFTER the cmp above (which we don't care about) and BEFORE
//     the jne executes on the next erun.
_ZF = 1
erun
jmp iat_dispatch

crc_hit:
// The first CRC guard to fire is the live one. Lock it and drop the siblings,
// otherwise a different guard could execute and poison the zero-flag read.
cmp crc_locked, 0
jne crc_read_zf
crc_location = eip
crc_locked = 1
bc
bp redirect_jne
bp crc_location
log "Locked CRC guard @ {p:crc_location}"

crc_read_zf:
sti                        // execute the test so ZF reflects the actual check
cmp _zf, 1
je iat_complete            // ZF set -> IAT is fully written
erun
jmp iat_dispatch


//============================================================================
// Phase 1e - IAT complete: keep only CreateThread, then run on to it
//============================================================================
iat_complete:
log "CRC zero flag caught - IAT should be fully written"
bc
bp CreateThread
erun                       // run on to the thread-creation site
jmp oep_walk_start


//----------------------------------------------------------------------------
// Fallback: no CRC guard was found. Force the redirect until CreateThread
// fires, exactly like the original CreateThread-signaled loop.
//----------------------------------------------------------------------------
no_crc_guard:
log "No CRC guard found - using CreateThread as the IAT-done signal"
bp CreateThread

force_zf_loop:
_ZF = 1
erun
cmp eip, redirect_jne
je force_zf_loop
// not the jne -> CreateThread fired -> IAT done, and we are already sitting on it


//============================================================================
// Phase 2 - from CreateThread, walk to the original entry point
//============================================================================
oep_walk_start:
log "At CreateThread - searching for OEP"
ct_entry = eip                  // CreateThread entry - used as a re-sync landmark

walk_from_ct:
watch_addr = [esp]              // return address into the caller of CreateThread
bp watch_addr
erun                           // run back up to that caller
rtr                            // return out of the caller

check_return:
cmp eip, ct_entry              // did we loop straight back into CreateThread?
je retry_walk                  // yes -> overshot, try again
bc watch_addr
sti

// OEP is reached via the SECOND indirect call ("FF D?") from here.
find_oep_call:
find eip, "FF D?"
cmp $result, 0
je oep_not_found
find $result + 0x1, "FF D?"
cmp $result, 0
je oep_not_found
watch_addr = $result
bp watch_addr
erun
cmp eip, watch_addr
je oep_found
// landed elsewhere -> fall through and retry the walk

retry_walk:
bc watch_addr
cmp eip, ct_entry
je walk_from_ct
erun
jmp walk_from_ct

oep_not_found:
bc watch_addr
erun
cmp eip, ct_entry
je walk_from_ct
msg "Couldn't find the OEP."
ret

oep_found:
bc
sti
cmt eip, "<--- OEP is here"
bph eip
msg "Found OEP @ {eip} and commented the address. :] booya"
ret


//============================================================================
// No redirect stubs matched
//============================================================================
no_stubs_found:
msg "No redirect stubs matched the pattern (scan gave up after 200 tries)."
ret
