//Written by hardlyprogramming
//Armadillo v6.xx - 9.xx : redirect NOP -> CRC/IAT wait -> OEP finder
//Run from entry point (or directly after the redirect finder).
//
//Structure:
//  1. Redirect finder  -> locate the "jne" redirect, NOP it       (unchanged)
//  2. CRC / IAT wait    -> lock the first CRC candidate, watch ZF  (rewrite)
//  3. OEP finder        -> find + comment the OEP                  (unchanged)


//===========================================================================
// 1. REDIRECT FINDER
//===========================================================================

bp VirtualProtect
erun
bc VirtualProtect
bp LoadLibraryA
erun
my_lib = eip

Back_In_Load_Lib:
rtr
sti
rtr

cmp eip, my_lib
je Back_In_Load_Lib
sti

//32bit
find cip, "E9 ?? ?? ?? ?? 83 B? ?? ?? ?? ?? 00"
cmp $result, 0
jne set_break

//64bit
//find cip, "E9 ?? ?? ?? ?? 48 83 B? ?? ?? ?? ?? ?? 00"
//cmp $result, 0
//jne set_break

erun
cmp eip, my_lib
je Back_In_Load_Lib
erun
jmp Back_In_Load_Lib

set_break:
possible_loc = $result
bp $result
erun
cmp eip, possible_loc
je found
bc possible_loc
cmp eip, my_lib
je Back_In_Load_Lib
erun
cmp eip, my_lib
je Back_In_Load_Lib
msg "Something went Wrong"
ret

found:
bc my_lib
msg "Found patch location at {p:possible_loc}.  NOPing the 'jne' instruction"
find cip, "75 ??"
cmp $result, 0
je ERROR_PATCHING
//saving the current bytes and address in order to restore later
bytes_to_restore = 2:[$result]
address_to_restore = $result
// patching the redirect with NOP
2:[$result] = 9090
log "address to restore = {p:address_to_restore}"

bc
bp address_to_restore


//===========================================================================
// 2. CRC / IAT WAIT
//    - find every "test / jne+5 / jmp" guard in the current page
//    - run: the FIRST one that fires after the patch is our target
//    - lock it, drop the rest, then watch ITS zero flag until it sets
//===========================================================================

//This searches for potential CRC check loops
findall mem.base(cip), "85 ?? 75 05 E9"

count = ref.count()
cmp count, 0
je no_crc_found                       // nothing to watch -> straight to OEP

// --- set a breakpoint on every candidate ---
i = 0
set_breakpoints:
    addr = ref.addr(i)
    bp addr
    log "CRC candidate {d:i} @ {p:addr}"
    i++
    cmp i, count
    jne set_breakpoints
log "{d:count} CRC candidate breakpoints set"

crc_location = 0

// --- Phase 1: run until the FIRST candidate fires ---
find_first_hit:
    erun
    hit = $breakpointexceptionaddress
    log "Break @ {p:hit}"

    i = 0
identify_hit:
    addr = ref.addr(i)
    cmp hit, addr
    je lock_crc
    i++
    cmp i, count
    jne identify_hit
    // hit was NOT a candidate (e.g. the patched jne bp) -> keep running
    jmp find_first_hit

lock_crc:
    crc_location = hit
    log "Locked CRC location @ {p:crc_location}"
    // Critical: drop ALL other breakpoints so only THIS address can fire.
    bc
    bp crc_location

// --- Phase 2: watch only crc_location, wait for ZF == 1 ---
watch_crc:
    erun                              // returns to crc_location (it loops)
    sti                               // execute the TEST so ZF sets
    cmp _zf, 1
    jne watch_crc

    log "Zero flag caught - IAT should be fully written"
    log "Restoring bytes"
restore_bytes:
    2:[address_to_restore] = bytes_to_restore
    // verify by read-back (memory assignment does not set $result) //always forget
    cmp 2:[address_to_restore], bytes_to_restore
    jne FAILED_RESTORE
    log "Bytes restored"
    bc
    log "Breakpoints cleared"
    msg "CRC handling finished"
    jmp Find_OEP

no_crc_found:
    log "No CRC pattern found - skipping to OEP finder"
    bc
    jmp Find_OEP


//===========================================================================
// 3. OEP FINDER  (Armadillo v6.xx - 9.xx)
//===========================================================================

Find_OEP:

bp CreateThread
erun
set my_1, eip

In_CreateThread:
my_2 = [esp]
bp [esp]
erun
rtr

Check_CreateThread:
cmp eip, my_1
je Missed_Break
bc my_2
sti

look_for_magic_call:
find eip, "FF D?"
cmp $result, 0
je Not_Found

find $result+1, "FF D?"
cmp $result, 0
je Not_Found
set my_2, $result
bp $result

erun
cmp eip, my_2
je Found_Oep


Missed_Break:
bc my_2
cmp eip, my_1
je In_CreateThread
erun
jmp In_CreateThread


Not_Found:
bc my_2
erun
cmp eip, my_1
je In_CreateThread
msg "Couldn't find the OEP"
ret


Found_Oep:
bc
sti
cmt eip, "<--- OEP is Here"
bph eip
msg "Found OEP @ {p:eip} and commented the address :]"
ret


//===========================================================================
// ERROR HANDLERS
//===========================================================================

FAILED_RESTORE:
msg "Failed when restoring bytes..."
ret

ERROR_PATCHING:
msg "Error while patching the 'jne'"
ret
